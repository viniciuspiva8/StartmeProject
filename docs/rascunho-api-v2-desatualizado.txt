const express = require('express');
const mysql = require('mysql2/promise');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(express.json());

const pool = mysql.createPool({
  host: 'localhost',
  user: 'root',
  password: '',  // Altere conforme sua senha
  database: 'mydb'
});

// Função utilitária
async function executeQuery(query, params, res) {
  try {
    const [result] = await pool.query(query, params);
    res.json(result);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
}

// --------- CRUD Usuarios ---------
app.post('/usuarios', (req, res) => {
  const { nome, telefone, email, tipo } = req.body;
  executeQuery('INSERT INTO Usuarios (Nome, Telefone, Email, Tipo) VALUES (?, ?, ?, ?)',
    [nome, telefone, email, tipo], res);
});

app.get('/usuarios', (req, res) => {
  executeQuery('SELECT * FROM Usuarios', [], res);
});

app.put('/usuarios/:id', (req, res) => {
  const { nome, telefone, email, tipo } = req.body;
  const { id } = req.params;
  executeQuery('UPDATE Usuarios SET Nome=?, Telefone=?, Email=?, Tipo=? WHERE Id_Usuario=?',
    [nome, telefone, email, tipo, id], res);
});

app.delete('/usuarios/:id', (req, res) => {
  const { id } = req.params;
  executeQuery('DELETE FROM Usuarios WHERE Id_Usuario=?', [id], res);
});
